## 接口契约

```python
# apps/render/base.py
from typing import Protocol, Literal
from dataclasses import dataclass, field
from pathlib import Path

NodeKind = Literal["claim", "figure", "table", "citation"]
EdgeKind = Literal["supports", "illustrates", "contradicts", "cites"]

@dataclass
class GraphNode:
    node_id: str          # claim:1 / figure:3 / table:5 / citation:42
    kind: NodeKind
    label: str            # 节点显示文字（claim 短摘要 / caption 头部 / bibkey）
    attrs: dict = field(default_factory=dict)

@dataclass
class GraphEdge:
    from_id: str
    to_id: str
    kind: EdgeKind
    label: str = ""

@dataclass
class PaperGraphModel:
    paper_arxiv_id: str
    nodes: list[GraphNode]
    edges: list[GraphEdge]

class GraphRenderer(Protocol):
    def render(self, graph: PaperGraphModel, out_dir: Path) -> Path: ...
```

## build_graph 规则（apps/render/graph.py）

```python
def build_graph(arxiv_id: str) -> PaperGraphModel:
    from apps.interpret.models import Claim, CounterSignal
    from apps.extract.models import Figure, Table, Citation

    nodes: list[GraphNode] = []
    edges: list[GraphEdge] = []
    cited_materials: dict[str, str] = {}   # full material_id → short

    claims = Claim.objects.filter(paper_arxiv_id=arxiv_id).prefetch_related("evidences")
    for c in claims:
        seq = c.claim_id.rsplit(":", 1)[1]
        node_id = f"claim:{seq}"
        nodes.append(GraphNode(
            node_id=node_id, kind="claim",
            label=c.text[:40] + ("…" if len(c.text) > 40 else ""),
            attrs={"claim_type": c.claim_type, "confidence": c.confidence,
                   "claim_id": c.claim_id},
        ))
        for e in c.evidences.all():
            kind = e.material_id.split(":")[1]   # figure / table / equation / section / citation
            if kind not in ("figure", "table", "citation"):
                continue   # equation / section 不入图（spec lock）
            short = f"{kind}:{e.material_id.rsplit(':', 1)[1]}"
            cited_materials[e.material_id] = short
            edge_kind = "illustrates" if e.relation == "illustrates" else "supports"
            edges.append(GraphEdge(from_id=node_id, to_id=short, kind=edge_kind))

    # counter_signal 简化为「contradicts」边（不作为独立节点）
    signals = CounterSignal.objects.filter(claim__paper_arxiv_id=arxiv_id).select_related("claim")
    for cs in signals:
        claim_seq = cs.claim.claim_id.rsplit(":", 1)[1]
        # contradicts 边从 claim 指向其反向 evidence material
        ev_kind = cs.evidence_material_id.split(":")[1]
        if ev_kind not in ("figure", "table", "citation"):
            continue
        short = f"{ev_kind}:{cs.evidence_material_id.rsplit(':', 1)[1]}"
        cited_materials[cs.evidence_material_id] = short
        edges.append(GraphEdge(
            from_id=f"claim:{claim_seq}",
            to_id=short,
            kind="contradicts",
            label=cs.signal_type,
        ))

    # 把被 cite 的 material 加成节点
    for full, short in cited_materials.items():
        kind = full.split(":")[1]
        if kind == "figure":
            f = Figure.objects.filter(material_id=full).first()
            if f:
                nodes.append(GraphNode(
                    node_id=short, kind="figure",
                    label=(f.caption or "")[:30],
                    attrs={"image_path": f.image_path, "page": f.page},
                ))
        elif kind == "table":
            t = Table.objects.filter(material_id=full).first()
            if t:
                nodes.append(GraphNode(
                    node_id=short, kind="table",
                    label=(t.caption or "")[:30],
                    attrs={"page": t.page},
                ))
        elif kind == "citation":
            c = Citation.objects.filter(material_id=full).first()
            if c:
                nodes.append(GraphNode(
                    node_id=short, kind="citation",
                    label=f"{c.bibkey} ({c.year or '?'})",
                    attrs={"raw_text": c.raw_text[:200]},
                ))

    return PaperGraphModel(arxiv_id, nodes, edges)
```

## ExcalidrawRenderer 实现要点

参考 [Excalidraw schema](https://github.com/excalidraw/excalidraw/blob/master/packages/excalidraw/data/types.ts):

最小可用 element 类型：
- `rectangle`（claim / table / citation 节点）
- `image`（figure 节点，base64 在 `files` 字段）
- `text`（节点内文字，作为独立 element 用 `containerId` 绑到 rectangle）
- `arrow`（边，含 `startBinding` / `endBinding`）

布局策略（简单分层）：
- y=100：claim 节点行（按 confidence 降序，间隔 280px）
- y=350：figure / table / citation 节点行（与对应 claim 列对齐 ± offset）
- y=600：citation 节点行（如果还有更多层级）
- arrow 自然垂直向下

边色：
- supports → `strokeColor: "#1971c2"` 蓝
- illustrates → `strokeColor: "#2f9e44"` 绿
- contradicts → `strokeColor: "#e03131"` 红 + `strokeStyle: "dashed"`
- cites → `strokeColor: "#868e96"` 灰

每个 element 的 id 用 nanoid（手写一个 16 字符随机生成，不引依赖）。

`files` 字段：
```python
{
  "<fileId>": {
    "id": "<fileId>",
    "mimeType": "image/png",
    "dataURL": "data:image/png;base64,<base64>",
    "created": <ms timestamp>,
  }
}
```

读 figure 节点的 `attrs["image_path"]` → base64 编码 → 写 dataURL。

## SVG fallback（apps/render/svg_renderer.py）

最简实现：
- `<svg>` viewbox 计算从节点坐标
- 节点：`<rect>` + `<text>`（claim/table/citation）；figure 用 `<image href="data:image/png;base64,...">`
- 边：`<line>` + `<text>`（边标签居中）

布局复用 ExcalidrawRenderer 同一份 `_layout(graph)` 函数（抽到 `apps/render/layout.py` 共用）。

## CLI

```python
# apps/render/management/commands/render_graph.py
class Command(BaseCommand):
    def add_arguments(self, parser):
        parser.add_argument("arxiv_id")
        parser.add_argument("--format", choices=["excalidraw", "svg"], default="excalidraw")
    def handle(self, *args, **opts):
        graph = build_graph(opts["arxiv_id"])
        out_dir = Path(settings.BASE_DIR) / "media" / "render" / opts["arxiv_id"]
        out_dir.mkdir(parents=True, exist_ok=True)
        renderer = ExcalidrawRenderer() if opts["format"] == "excalidraw" else SvgRenderer()
        path = renderer.render(graph, out_dir)
        artifact = persist_artifact(opts["arxiv_id"], opts["format"], path)
        self.stdout.write(f"[render] done: {artifact.artifact_id} → {path}")
```
