```python
# apps/interpret/base.py
from typing import Protocol, Literal
from dataclasses import dataclass, field

ClaimType = Literal["proposal", "result", "ablation", "theoretical"]
EvidenceRelation = Literal["illustrates", "quantifies", "supports"]
SignalType = Literal["limitation", "ablation_drop", "failure_case", "prior_work_better"]

@dataclass
class ClaimEvidenceDraft:
    material_id: str           # 必须能在 extract_* 表查到
    relation: EvidenceRelation = "supports"

@dataclass
class ClaimDraft:
    text: str                  # 中文化短陈述
    text_en: str = ""          # 原文 span（可空）
    claim_type: ClaimType = "result"
    source_section_path: str = ""
    confidence: float = 0.0    # LLM 自评
    evidences: list[ClaimEvidenceDraft] = field(default_factory=list)

@dataclass
class CounterSignalDraft:
    text: str
    signal_type: SignalType
    evidence_material_id: str  # 必填
    # claim_idx 在 InterpretResult 内通过位置挂回
    claim_idx: int             # 指向 InterpretResult.claims 的索引

@dataclass
class InterpretResult:
    paper_arxiv_id: str
    claims: list[ClaimDraft]
    counter_signals: list[CounterSignalDraft]

class Interpreter(Protocol):
    def interpret(self, paper_arxiv_id: str, pdf_path: Path) -> InterpretResult: ...
```

## Material Catalog 构造（apps/interpret/catalog.py）

读 `extract_*` 表组装 LLM cite 目标清单：

```python
def build_catalog(arxiv_id: str) -> tuple[str, set[str]]:
    """返回 (catalog_text, allowed_material_ids).

    catalog_text 给 LLM 看，allowed_material_ids 用于 cite 校验。"""
    from apps.extract.models import Section, Figure, Table, Equation, Citation
    lines = []
    allowed = set()
    # 只用 level=1 的 canonical sections（清洁层已 demote noise）
    for s in Section.objects.filter(paper_arxiv_id=arxiv_id, level=1).order_by("material_id"):
        lines.append(f"[§:{_seq(s.material_id)}] {s.path}")
        allowed.add(s.material_id)
    for f in Figure.objects.filter(paper_arxiv_id=arxiv_id).order_by("material_id"):
        cap = (f.caption or "")[:200]
        lines.append(f"[fig:{_seq(f.material_id)}] p{f.page} {cap}")
        allowed.add(f.material_id)
    for t in Table.objects.filter(paper_arxiv_id=arxiv_id).order_by("material_id"):
        cap = (t.caption or "")[:200]
        lines.append(f"[tbl:{_seq(t.material_id)}] p{t.page} {cap}")
        allowed.add(t.material_id)
    for e in Equation.objects.filter(paper_arxiv_id=arxiv_id).order_by("material_id")[:20]:
        latex = (e.latex_or_text or "")[:120]
        lines.append(f"[eq:{_seq(e.material_id)}] {latex}")
        allowed.add(e.material_id)
    return "\n".join(lines), allowed
```

LLM 看到的 cite 形式是 `[fig:1]` 之类的简写；返回时**也用简写**，由后处理映射回 full material_id（`<arxiv_id>:figure:1`）。简写 → full 的映射要严格校验，未在 allowed 集合中的直接丢。

## Paper Markdown 输入

调 `apps.extract.extractors.docling_ext._convert()` 复用 paper-level 缓存：

```python
def get_paper_markdown(arxiv_id: str, pdf_path: Path) -> str:
    from apps.extract.extractors.docling_ext import _convert
    doc = _convert(arxiv_id, pdf_path)
    return doc.export_to_markdown()
```

**长论文 token 控制**：若 markdown 超过预算（如 30K chars），按 section 切块（`re.split(r"\n## ", md)`），每块单独抽 claims 后合并。本 ft 简化：直接截到 30K chars，不分块（leworldmodel / convnextv2 都在阈值内）。

```python
class Command(BaseCommand):
    def add_arguments(self, parser):
        parser.add_argument("arxiv_id")
        parser.add_argument("--pdf-path", default=None)

        arxiv_id = opts["arxiv_id"]
```
