## 关键组件设计

### ClaimCard（核心）

```tsx
import { BlockMath } from "react-katex";
import "katex/dist/katex.min.css";

export interface Claim {
  claim_id: string;
  text: string;
  claim_type: "proposal" | "result" | "ablation" | "theoretical";
  confidence: number;
  evidences: Array<{ kind: "figure"|"table"|"section"; ref_id: string; label: string; image_path?: string }>;
  equation_evidences: Array<{ ref_id: string; label: string }>;
  counter_signals: Array<{ text: string; signal_type: string; evidence_ref_id: string }>;
}

export function ClaimCard({ claim, onCiteClick }: { claim: Claim; onCiteClick: (refId: string) => void }) {
  const typeColor = {
    proposal:    "bg-blue-100 text-blue-800",
    result:      "bg-green-100 text-green-800",
    ablation:    "bg-orange-100 text-orange-800",
    theoretical: "bg-purple-100 text-purple-800",
  }[claim.claim_type];

  return (
    <Card className="p-4 space-y-3">
      <div className="flex items-center gap-2 text-xs">
        <span className={`px-2 py-0.5 rounded ${typeColor}`}>{claim.claim_type}</span>
        <span className="text-muted-foreground">conf {claim.confidence.toFixed(2)}</span>
      </div>
      <p className="text-sm leading-relaxed">{claim.text}</p>
      {claim.equation_evidences.map(eq => (
        <div key={eq.ref_id} className="bg-slate-50 rounded p-2 overflow-x-auto">
          <EquationBlock latex={eq.label} />
        </div>
      ))}
      <div className="flex flex-wrap gap-1">
        {claim.evidences.map(ev => (
          <button
            key={ev.ref_id}
            onClick={() => onCiteClick(ev.ref_id)}
            className="text-xs px-2 py-0.5 rounded bg-slate-200 hover:bg-slate-300 transition"
            title={ev.label}
          >
            {ev.kind === "figure" ? "📊" : ev.kind === "table" ? "📋" : "§"} {ev.ref_id}
          </button>
        ))}
      </div>
      {claim.counter_signals.map((cs, i) => (
        <div key={i} className="border-l-4 border-red-400 bg-red-50 p-2 text-xs">
          <span className="font-semibold text-red-700">⚠ {cs.signal_type}</span> {cs.text}
        </div>
      ))}
    </Card>
  );
}
```

### EquationBlock（KaTeX 渲染兜底）

```tsx
import { BlockMath } from "react-katex";

export function EquationBlock({ latex }: { latex: string }) {
  try {
    return <BlockMath math={sanitizeLatex(latex)} errorColor="#cc0000" />;
  } catch {
    return <code className="text-xs">{latex}</code>;
  }
}

function sanitizeLatex(s: string): string {
  // mathtext 那套预处理：\colon → :, \text → \mathrm 等
  return s.replace(/\\colon/g, ":").replace(/\\text\{([^}]*)\}/g, "\\mathrm{$1}");
}
```
